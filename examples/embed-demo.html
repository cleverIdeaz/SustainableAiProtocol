🔧 Integration Example

// Fetch MCP telemetry
fetch('http://localhost:3001/mcp/telemetry')
  .then(res => res.json())
  .then(data => {
    const stats = JSON.parse(data.contents[0].text);
    console.log('Global stats:', stats.global_stats);
    console.log('Recent events:', stats.recent_events);
  });
