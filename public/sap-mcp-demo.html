What is MCP?

Model Context Protocol (MCP)

MCP is an open-source standard for connecting AI applications to external systems and data sources. It enables AI models to access real-time information, perform actions, and interact with external APIs.

SAP extends MCP with a sustainability layer, providing environmental context and optimization capabilities to every AI interaction.

Integration Examples

Python MCP Client

from mcp import ClientSession, StdioServerParameters
from mcp.client.stdio import stdio_client

async def main():
    async with stdio_client(StdioServerParameters(
        command="node",
        args=["sap-mcp-server.js"]
    )) as (read, write):
        async with ClientSession(read, write) as session:
            # Get environmental context
            result = await session.call_tool(
                "get_environmental_context",
                {"location": "global"}
            )
            
            # Optimize prompt for sustainability
            optimized = await session.call_tool(
                "optimize_prompt",
                {
                    "prompt": "What is sustainable AI?",
                    "model": "gpt-4",
                    "preferences": "energy-focused"
                }
            )
            
            print(f"Environmental context: {result}")
            print(f"Optimized prompt: {optimized}")

if __name__ == "__main__":
    import asyncio
    asyncio.run(main())

JavaScript MCP Client

const { Client } = require('@modelcontextprotocol/sdk/client/index.js');
const { StdioClientTransport } = require('@modelcontextprotocol/sdk/client/stdio.js');

async function runSAPMCP() {
    const transport = new StdioClientTransport({
        command: 'node',
        args: ['sap-mcp-server.js']
    });
    
    const client = new Client({
        name: 'sap-client',
        version: '1.0.0'
    }, {
        capabilities: {}
    });
    
    await client.connect(transport);
    
    // Get environmental data
    const envData = await client.callTool({
        name: 'get_environmental_context',
        arguments: { location: 'global' }
    });
    
    // Optimize prompt
    const optimized = await client.callTool({
        name: 'optimize_prompt',
        arguments: {
            prompt: 'What is sustainable AI?',
            model: 'gpt-4',
            preferences: 'energy-focused'
        }
    });
    
    console.log('Environmental data:', envData);
    console.log('Optimized prompt:', optimized);
    
    await client.close();
}

runSAPMCP().catch(console.error);
