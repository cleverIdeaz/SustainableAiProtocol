JavaScript Integration

// SAP Tracking Function
function trackSAPInteraction() {
    // Increment local counter
    const counter = parseInt(localStorage.getItem('sap_counter') || '0') + 1;
    localStorage.setItem('sap_counter', counter);
    
    // Update display
    document.querySelector('.sap-button text').textContent = counter;
    
    // Send to SAP API
    fetch('/api/track', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
            prompt: 'User interaction',
            model: 'unknown',
            tokens: 1,
            userId: localStorage.getItem('sap_user_id') || 'anonymous'
        })
    });
}
